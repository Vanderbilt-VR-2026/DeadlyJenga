Shader "Candy Tropical/Avatar Health"
{
 Properties {_BaseMap("Coat",2D)="white"{} [Normal] _BumpMap("Fine normal",2D)="bump"{} _BumpScale("Normal strength",Float)=0 _BaseColor("Candy color",Color)=(1,.2,.1,1) _Smoothness("Gloss",Range(0,1))=.8 _Metallic("Metal",Range(0,1))=0 [HDR] _EmissionColor("Emission",Color)=(0,0,0,1) _Enemy("Enemy fill",Float)=0 _Health("Health",Float)=1 _Bottom("Feet",Float)=0 _Top("Head",Float)=2 _MeltStart("Melt start",Float)=-10000 _Sink("Sink progress",Float)=0 _Recovering("Recovering",Float)=0}
 SubShader {Tags {"RenderPipeline"="UniversalPipeline" "RenderType"="Opaque"}
 Pass {Tags {"LightMode"="UniversalForward"} HLSLPROGRAM
 #pragma vertex vert
 #pragma fragment frag
 #pragma multi_compile_instancing
 #pragma multi_compile _ _MAIN_LIGHT_SHADOWS _MAIN_LIGHT_SHADOWS_CASCADE
 #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
 #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Lighting.hlsl"
 #include "Packages/com.unity.render-pipelines.core/ShaderLibrary/Packing.hlsl"
 #include "Packages/com.unity.render-pipelines.core/ShaderLibrary/CommonMaterial.hlsl"
 TEXTURE2D(_BaseMap);SAMPLER(sampler_BaseMap);TEXTURE2D(_BumpMap);SAMPLER(sampler_BumpMap);
 CBUFFER_START(UnityPerMaterial)
 float4 _BaseColor,_EmissionColor,_BaseMap_ST;float _BumpScale;float _Metallic,_Smoothness,_Enemy,_Health,_Bottom,_Top,_MeltStart,_Sink,_Recovering;
 CBUFFER_END
 struct A{float4 positionOS:POSITION;float3 normalOS:NORMAL;float4 tangentOS:TANGENT;float2 uv:TEXCOORD0;UNITY_VERTEX_INPUT_INSTANCE_ID};struct V{float4 positionCS:SV_POSITION;float3 world:TEXCOORD0;float3 normal:TEXCOORD1;float melt:TEXCOORD2;float4 tangent:TEXCOORD3;float2 uv:TEXCOORD4;UNITY_VERTEX_OUTPUT_STEREO};
 V vert(A i){V o;UNITY_SETUP_INSTANCE_ID(i);UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);float3 p=TransformObjectToWorld(i.positionOS.xyz);float age=_Time.y-_MeltStart;float melt=_MeltStart<0?0:saturate(age/.65);float3 center=TransformObjectToWorld(float3(0,0,0));p.xz=center.xz+(p.xz-center.xz)*(1+melt*.65);p.y=lerp(p.y,_Bottom+.018,melt)-saturate(_Sink)*max(.1,_Top-_Bottom)*.8;o.world=p;o.positionCS=TransformWorldToHClip(p);o.normal=normalize(lerp(TransformObjectToWorldNormal(i.normalOS),float3(0,1,0),melt));o.melt=_MeltStart<0?0:saturate((age-.75)/.7);o.tangent=float4(TransformObjectToWorldDir(i.tangentOS.xyz),i.tangentOS.w*GetOddNegativeScale());o.uv=TRANSFORM_TEX(i.uv,_BaseMap);return o;}
 float hash(float2 p){return frac(sin(dot(p,float2(127.1,311.7)))*43758.5453);}
 float noise(float2 p){float2 a=floor(p),b=frac(p);b=b*b*(3-2*b);return lerp(lerp(hash(a),hash(a+float2(1,0)),b.x),lerp(hash(a+float2(0,1)),hash(a+1),b.x),b.y);}
 half4 frag(V i):SV_Target{UNITY_SETUP_STEREO_EYE_INDEX_POST_VERTEX(i);float organic=noise(i.world.xz*4)*.7+noise(i.world.xz*9+3)*.3;clip(organic-i.melt);float level=(_Top-_Bottom)*_Health+_Bottom;float h=_Health>=.9999?0:(_Health<=.0001?1:smoothstep(level-.015,level+.015,i.world.y));float3 pigment=_BaseColor.rgb*SAMPLE_TEXTURE2D(_BaseMap,sampler_BaseMap,i.uv).rgb;float enemyLevel=_Bottom+(_Top-_Bottom)*(1-_Health);float enemyFill=_Health>=.9999?0:_Health<=.0001?1:1-smoothstep(enemyLevel-.015,enemyLevel+.015,i.world.y);float3 c=_Enemy>.5?lerp(pigment,float3(.35,.37,.39),enemyFill):lerp(pigment,lerp(pigment,float3(1,1,1),.48),h);Light light=GetMainLight(TransformWorldToShadowCoord(i.world));float3 n=normalize(i.normal),v=normalize(_WorldSpaceCameraPos-i.world);if(_BumpScale>.001&&dot(i.tangent.xyz,i.tangent.xyz)>.01){float3 t=normalize(i.tangent.xyz);float3 normalTS=UnpackNormalScale(SAMPLE_TEXTURE2D(_BumpMap,sampler_BumpMap,i.uv),_BumpScale);n=normalize(t*normalTS.x+cross(n,t)*i.tangent.w*normalTS.y+n*normalTS.z);}float spec=pow(saturate(dot(n,normalize(light.direction+v))),lerp(24,180,_Smoothness));float width=max(.012,(_Top-_Bottom)*.006);float rim=(1-smoothstep(width*.35,width,abs(i.world.y-level)))*step(.001,_Health)*(1-step(.999,_Health));float3 bright=_BaseColor.rgb/max(.12,max(_BaseColor.r,max(_BaseColor.g,_BaseColor.b)));float pulse=1+_Recovering*.2*sin(_Time.y*6);float metal=saturate(_Metallic);float3 specular=lerp(float3(.7,.7,.7),c*1.4,metal);float3 reflection=GlossyEnvironmentReflection(reflect(-v,n),i.world,1-saturate(_Smoothness),1);float3 surface=c*(SampleSH(n)+light.color*saturate(dot(n,light.direction))*light.shadowAttenuation)*(1-metal*.65)+light.color*spec*specular*light.shadowAttenuation;surface+=reflection*lerp(float3(.035,.035,.035),c,metal)*saturate(_Smoothness);return half4(surface+_EmissionColor.rgb+bright*rim*pulse,1);}
 ENDHLSL }
